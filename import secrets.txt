import secrets
import string
import tkinter as tk
from tkinter import messagebox
import pyperclip


# Store last 5 passwords only during this session
history = []


def generate_password():
    try:
        length = int(length_var.get())

        if length < 8:
            messagebox.showerror(
                "Invalid Length",
                "Password length must be at least 8 characters."
            )
            return

        selected = []

        if uppercase_var.get():
            selected.append(string.ascii_uppercase)

        if lowercase_var.get():
            selected.append(string.ascii_lowercase)

        if numbers_var.get():
            selected.append(string.digits)

        if symbols_var.get():
            selected.append(string.punctuation)

        if len(selected) < 2:
            messagebox.showerror(
                "Invalid Selection",
                "Please select at least two character types."
            )
            return

        # Remove ambiguous characters if requested
        if ambiguous_var.get():
            selected = [
                chars.translate(
                    str.maketrans("", "", "0Oo1lI")
                )
                for chars in selected
            ]

        # Make sure each selected type is represented
        password_characters = [
            secrets.choice(chars) for chars in selected
        ]

        # Create the remaining characters
        all_characters = "".join(selected)

        while len(password_characters) < length:
            password_characters.append(
                secrets.choice(all_characters)
            )

        # Securely shuffle the password
        for i in range(len(password_characters) - 1, 0, -1):
            j = secrets.randbelow(i + 1)
            password_characters[i], password_characters[j] = (
                password_characters[j],
                password_characters[i]
            )

        password = "".join(password_characters)

        password_var.set(password)

        # Strength
        if length >= 16 and len(selected) >= 4:
            strength = "STRONG"
        elif length >= 12 and len(selected) >= 3:
            strength = "MEDIUM"
        else:
            strength = "WEAK"

        strength_var.set("Strength: " + strength)

        # Keep only last 5 passwords
        history.append(password)

        if len(history) > 5:
            history.pop(0)

        update_history()

        # Automatically copy
        try:
            pyperclip.copy(password)
            clipboard_var.set("Password copied to clipboard!")
        except Exception:
            clipboard_var.set("Password generated.")

    except ValueError:
        messagebox.showerror(
            "Invalid Input",
            "Please enter a valid password length."
        )


def update_history():
    history_text.delete("1.0", tk.END)

    for number, password in enumerate(history, start=1):
        history_text.insert(
            tk.END,
            f"{number}. {password}\n"
        )


def copy_password():
    password = password_var.get()

    if password:
        try:
            pyperclip.copy(password)
            clipboard_var.set("Password copied to clipboard!")
        except Exception:
            messagebox.showerror(
                "Clipboard Error",
                "Could not copy the password."
            )
    else:
        messagebox.showwarning(
            "No Password",
            "Generate a password first."
        )


def clear_all():
    password_var.set("")
    strength_var.set("")
    clipboard_var.set("")

    uppercase_var.set(True)
    lowercase_var.set(True)
    numbers_var.set(True)
    symbols_var.set(True)
    ambiguous_var.set(False)

    history.clear()
    update_history()

# MAIN WINDOW
# ==============================

root.title("Secure Password Generator")
root.geometry("500x650")
root.resizable(False, False)

title = tk.Label(
    root,
    text="SECURE PASSWORD GENERATOR",
    font=("Arial", 18, "bold")
)

# Password length
tk.Label(
    root,
    text="Password Length:",
    font=("Arial", 12)
).pack()

length_spinbox = tk.Spinbox(
    root,
    from_=8,
    to=128,
    textvariable=length_var,
    width=10,
    font=("Arial", 12)
)
length_spinbox.pack(pady=10)


# Character selections
tk.Label(
    root,
    text="Character Types:",
    font=("Arial", 12, "bold")
).pack(pady=5)

uppercase_var = tk.BooleanVar(value=True)
lowercase_var = tk.BooleanVar(value=True)
numbers_var = tk.BooleanVar(value=True)
symbols_var = tk.BooleanVar(value=True)

tk.Checkbutton(
    root,
    text="Uppercase Letters",
    variable=uppercase_var
).pack(anchor="w", padx=120)

tk.Checkbutton(
    root,
    text="Lowercase Letters",
    variable=lowercase_var
).pack(anchor="w", padx=120)

    root,
    text="Numbers",
    variable=numbers_var
).pack(anchor="w", padx=120)

tk.Checkbutton(
    root,
    text="Symbols",
    variable=symbols_var
).pack(anchor="w", padx=120)


# Ambiguous characters

tk.Checkbutton(
    root,
    text="Exclude ambiguous characters (0, O, l, 1, I)",
    variable=ambiguous_var
).pack(pady=15)


# Generate button
tk.Button(
    root,
    text="Generate Password",
    command=generate_password,
    font=("Arial", 12, "bold"),
    width=25
).pack(pady=10)


# Password display
password_var = tk.StringVar()

tk.Label(
    root,
    text="Generated Password:",
    font=("Arial", 11, "bold")
).pack(pady=5)

    root,
    textvariable=password_var,
    font=("Arial", 12),
    width=38,
    justify="center"

# Strength
strength_var = tk.StringVar()

tk.Label(
    root,
    textvariable=strength_var,
    font=("Arial", 13, "bold")
).pack(pady=10)


# Copy button
tk.Button(
    root,
    text="Copy to Clipboard",
    command=copy_password,
    width=25
).pack(pady=5)


clipboard_var = tk.StringVar()

tk.Label(
    root,
    textvariable=clipboard_var
).pack()


# History
tk.Label(
    root,
    text="Last 5 Generated Passwords:",
    font=("Arial", 11, "bold")
).pack(pady=10)

history_text = tk.Text(
    root,
    height=6,
    width=45
)
history_text.pack()


# Clear button

    root,
    text="Clear",

    width=25
).pack(pady=15)
